function data
    asm "section .data"
    asm "msg_hello:   db 'Mini DSL echo demo',10,0"
    asm "msg_prompt:  db 'Type something and press ENTER:',10,0"
    asm "msg_prefix:  db 'You typed: ',0"
    asm "newline:     db 10,0"
end

array input_buf 64
array input_len 8

function print_str
    asm "push rdi"
    asm "push rax"
    asm "push rdx"

    asm "mov rdx, 0"
    asm ".len_loop:"
    asm "cmp byte [rsi+rdx], 0"
    asm "je .len_done"
    asm "inc rdx"
    asm "jmp .len_loop"
    asm ".len_done:"

    asm "mov rax, 1"
    asm "mov rdi, 1"
    asm "syscall"

    asm "pop rdx"
    asm "pop rax"
    asm "pop rdi"
end

function print_newline
    asm "lea rsi, [rel newline]"
    execute print_str
end

function read_line
    asm "mov rax, 0"
    asm "mov rdi, 0"
    asm "lea rsi, [rel input_buf]"
    asm "mov rdx, 63"
    asm "syscall"

    asm "cmp rax, 0"
    asm "jle .rl_done"

    asm "mov [rel input_len], rax"
    asm "mov rcx, rax"
    asm "dec rcx"
    asm "cmp byte [rel input_buf+rcx], 10"
    asm "jne .rl_done"
    asm "mov byte [rel input_buf+rcx], 0"
    asm "mov [rel input_len], rcx"

    asm ".rl_done:"
end

function main
    asm "lea rsi, [rel msg_hello]"
    execute print_str

    asm "lea rsi, [rel msg_prompt]"
    execute print_str

    execute read_line

    asm "lea rsi, [rel msg_prefix]"
    execute print_str

    asm "lea rsi, [rel input_buf]"
    execute print_str

    execute print_newline
end

function _start
    asm "global _start"
    asm "_start:"
    execute main
    asm "mov rax, 60"
    asm "xor rdi, rdi"
    asm "syscall"
end
